## 7.1 RBAC权限模型

### 7.1.1 权限体系设计

本系统采用基于角色的访问控制（RBAC）模型，包含用户（User）、角色（Role）、权限（Permission）三个核心实体。

**权限层级结构：**

```
系统权限
├── 用户管理
│   ├── users:view（查看用户）
│   ├── users:create（创建用户）
│   ├── users:edit（编辑用户）
│   ├── users:delete（删除用户）
│   └── users:ban（封禁用户）
├── 商品管理
│   ├── items:view（查看商品）
│   ├── items:create（发布商品）
│   ├── items:edit（编辑商品）
│   ├── items:delete（删除商品）
│   └── items:audit（审核商品）
├── 交易管理
│   ├── transactions:view（查看交易）
│   ├── transactions:manage（管理交易）
│   └── transactions:refund（退款处理）
├── 系统管理
│   ├── system:config（系统配置）
│   ├── system:logs（查看日志）
│   └── system:sync（同步管理）
└── 内容管理
    ├── content:moderate（内容审核）
    └── content:report（举报处理）
```

### 7.1.2 权限检查实现

#### 后端权限装饰器

```python
# core/security.py
from functools import wraps
from fastapi import HTTPException, status
from typing import List

def require_permissions(*required_permissions: str):
    """权限检查装饰器"""
    
    def decorator(func):
        @wraps(func)
        async def wrapper(*args, **kwargs):
            # 获取当前用户
            current_user = kwargs.get('current_user')
            if not current_user:
                raise HTTPException(
                    status_code=status.HTTP_401_UNAUTHORIZED,
                    detail="Authentication required"
                )
            
            # 查询用户权限
            user_permissions = get_user_permissions(current_user.id)
            
            # 检查是否拥有所需权限
            has_permission = all(
                perm in user_permissions 
                for perm in required_permissions
            )
            
            if not has_permission:
                raise HTTPException(
                    status_code=status.HTTP_403_FORBIDDEN,
                    detail=f"Missing permissions: {', '.join(required_permissions)}"
                )
            
            return await func(*args, **kwargs)
        
        return wrapper
    return decorator

def get_user_permissions(user_id: int) -> List[str]:
    """获取用户的所有权限"""
    
    with db_manager.session_scope("mysql") as session:
        permissions = session.execute(
            select(Permission.permission_code)
            .join(RolePermission)
            .join(UserRole)
            .where(UserRole.user_id == user_id)
        ).scalars().all()
        
    return list(permissions)
```

#### 资源所有权检查

```python
def check_ownership(resource_type: str):
    """检查资源所有权"""
    
    async def ownership_checker(
        resource_id: int,
        current_user: User = Depends(get_current_user)
    ):
        with db_manager.session_scope("mysql") as session:
            if resource_type == "item":
                resource = session.get(Item, resource_id)
                if not resource or resource.seller_id != current_user.id:
                    raise HTTPException(
                        status_code=status.HTTP_403_FORBIDDEN,
                        detail="You don't own this item"
                    )
            elif resource_type == "transaction":
                resource = session.get(Transaction, resource_id)
                if not resource or (
                    resource.buyer_id != current_user.id and 
                    resource.seller_id != current_user.id
                ):
                    raise HTTPException(
                        status_code=status.HTTP_403_FORBIDDEN,
                        detail="You are not part of this transaction"
                    )
        
        return resource
    
    return ownership_checker
```

## 7.2 身份认证

### 7.2.1 JWT令牌认证

```python
# core/security.py
from jose import JWTError, jwt
from passlib.context import CryptContext
from datetime import datetime, timedelta

pwd_context = CryptContext(schemes=["bcrypt"], deprecated="auto")

def create_access_token(data: dict, expires_delta: timedelta = None):
    """创建访问令牌"""
    
    to_encode = data.copy()
    
    if expires_delta:
        expire = datetime.utcnow() + expires_delta
    else:
        expire = datetime.utcnow() + timedelta(minutes=15)
    
    to_encode.update({"exp": expire, "iat": datetime.utcnow()})
    
    encoded_jwt = jwt.encode(
        to_encode, 
        settings.SECRET_KEY, 
        algorithm=settings.ALGORITHM
    )
    
    return encoded_jwt

def verify_password(plain_password: str, hashed_password: str) -> bool:
    """验证密码"""
    return pwd_context.verify(plain_password, hashed_password)

def get_password_hash(password: str) -> str:
    """密码哈希"""
    return pwd_context.hash(password)

# 登录接口
@router.post("/auth/login")
async def login(credentials: LoginRequest):
    """用户登录"""
    
    with db_manager.session_scope("mysql") as session:
        user = session.execute(
            select(User).where(User.username == credentials.username)
        ).scalar_one_or_none()
        
        if not user or not verify_password(credentials.password, user.password_hash):
            raise HTTPException(
                status_code=status.HTTP_401_UNAUTHORIZED,
                detail="Incorrect username or password"
            )
        
        if not user.is_active:
            raise HTTPException(
                status_code=status.HTTP_403_FORBIDDEN,
                detail="Account is disabled"
            )
        
        # 创建访问令牌
        access_token = create_access_token(
            data={"sub": str(user.id), "username": user.username},
            expires_delta=timedelta(days=7)
        )
        
        # 创建刷新令牌
        refresh_token = create_access_token(
            data={"sub": str(user.id), "type": "refresh"},
            expires_delta=timedelta(days=30)
        )
        
        # 更新最后登录时间
        user.last_login_at = datetime.utcnow()
        session.commit()
        
        # 记录审计日志
        log_audit(user.id, "LOGIN", "auth", user.id, "User logged in")
    
    return {
        "access_token": access_token,
        "refresh_token": refresh_token,
        "token_type": "bearer",
        "expires_in": 604800  # 7天
    }
```

### 7.2.2 刷新令牌机制

```python
@router.post("/auth/refresh")
async def refresh_token(refresh_request: RefreshTokenRequest):
    """刷新访问令牌"""
    
    try:
        payload = jwt.decode(
            refresh_request.refresh_token,
            settings.SECRET_KEY,
            algorithms=[settings.ALGORITHM]
        )
        
        if payload.get("type") != "refresh":
            raise HTTPException(status_code=401, detail="Invalid token type")
        
        user_id = int(payload.get("sub"))
        
        # 验证用户状态
        with db_manager.session_scope("mysql") as session:
            user = session.get(User, user_id)
            if not user or not user.is_active:
                raise HTTPException(status_code=401, detail="User not found or inactive")
        
        # 创建新的访问令牌
        new_access_token = create_access_token(
            data={"sub": str(user_id), "username": user.username},
            expires_delta=timedelta(days=7)
        )
        
        return {
            "access_token": new_access_token,
            "token_type": "bearer"
        }
        
    except JWTError:
        raise HTTPException(status_code=401, detail="Invalid refresh token")
```

### 7.3.3 敏感数据加密

```python
from cryptography.fernet import Fernet

class DataEncryption:
    """数据加密工具"""
    
    def __init__(self, key: bytes):
        self.cipher = Fernet(key)
    
    def encrypt(self, data: str) -> str:
        """加密数据"""
        return self.cipher.encrypt(data.encode()).decode()
    
    def decrypt(self, encrypted_data: str) -> str:
        """解密数据"""
        return self.cipher.decrypt(encrypted_data.encode()).decode()

# 加密用户手机号
encryptor = DataEncryption(settings.ENCRYPTION_KEY)

    user = get_user(user_id)
```

```python
# 审计日志装饰器
def audit_log(action: str, resource: str):
    """审计日志装饰器"""
    
    def decorator(func):
        @wraps(func)
        async def wrapper(*args, **kwargs):
            current_user = kwargs.get('current_user')
            request = kwargs.get('request')
            
            # 执行原函数
            result = await func(*args, **kwargs)
            
            # 记录审计日志
            with db_manager.session_scope("mysql") as session:
                log = AuditLog(
                    user_id=current_user.id if current_user else None,
                    action=action,
                    table_name=resource,
                    record_id=result.get('id') if isinstance(result, dict) else None,
                    details=json.dumps(kwargs),
                    ip_address=request.client.host if request else None,
                    created_at=datetime.utcnow()
                )
                session.add(log)
                session.commit()
            
            return result
        
        return wrapper
    return decorator
```

### 7.4.2 审计日志查询

```sql
-- 查询用户操作历史
SELECT 
    al.id,
    al.action,
    al.table_name,
    al.record_id,
    al.details,
    al.ip_address,
    al.created_at,
    u.username
FROM audit_logs al
LEFT JOIN users u ON al.user_id = u.id
WHERE al.user_id = :user_id
  AND al.created_at >= DATE_SUB(NOW(), INTERVAL 30 DAY)
ORDER BY al.created_at DESC
LIMIT 100;

-- 查询敏感操作
SELECT 
    al.*,
    u.username
FROM audit_logs al
LEFT JOIN users u ON al.user_id = u.id
WHERE al.action IN ('DELETE', 'PASSWORD_CHANGE', 'ACCOUNT_BANNED')
  AND al.created_at >= DATE_SUB(NOW(), INTERVAL 7 DAY)
ORDER BY al.created_at DESC;
```
