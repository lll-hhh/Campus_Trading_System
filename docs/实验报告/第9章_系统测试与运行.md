## 9.4 性能测试

性能测试评估系统在不同负载下的表现,包括响应时间、吞吐量、资源占用等指标。通过并发测试、压力测试、负载测试识别性能瓶颈,为优化提供数据支撑。

### 9.4.1 并发测试

并发测试模拟多用户同时访问系统,验证系统的并发处理能力和数据一致性。重点测试高并发场景下的数据库连接池、事务隔离、锁竞争等问题。

**测试目标：**
- 验证100并发用户下系统正常运行
- 响应时间P95 < 200ms
- 成功率 > 99%
- 无数据竞争导致的数据不一致

**测试实现（基于aiohttp的异步并发）：**

```python
# tests/test_performance.py
import asyncio
import aiohttp
import time
from typing import List, Dict
from statistics import mean, median
import pytest

class PerformanceTestSuite:
    """
    性能测试套件
    
    使用asyncio + aiohttp实现高并发请求测试
    支持多种请求模式（GET/POST/PUT/DELETE）
    收集详细的性能指标（响应时间、成功率、吞吐量）
    """
    
    def __init__(self, base_url: str):
        self.base_url = base_url
        self.results: List[Dict] = []
    
    async def send_request(
        self, 
        session: aiohttp.ClientSession,
        method: str,
        endpoint: str,
        **kwargs
    ) -> Dict:
        """
        发送HTTP请求并记录性能指标
        
        返回：包含状态码、响应时间、是否成功的字典
        """
        url = f"{self.base_url}{endpoint}"
        start_time = time.time()
        
        try:
            async with session.request(method, url, **kwargs) as response:
                await response.read()  # 读取响应体
                duration = time.time() - start_time
                
                return {
                    "status": response.status,
                    "duration_ms": duration * 1000,
                    "success": 200 <= response.status < 300,
                    "error": None
                }
        except Exception as e:
            duration = time.time() - start_time
            return {
                "status": 0,
                "duration_ms": duration * 1000,
                "success": False,
                "error": str(e)
            }
    
    async def concurrent_requests(
        self,
        method: str,
        endpoint: str,
        num_requests: int,
        **kwargs
    ) -> Dict:
        """
        并发发送多个请求
        
        参数：
            method: HTTP方法（GET/POST等）
            endpoint: API端点
            num_requests: 并发请求数
            **kwargs: 请求参数（headers, json, params等）
        
        返回：
            性能统计数据
        """
        async with aiohttp.ClientSession() as session:
            # 创建并发任务
            tasks = [
                self.send_request(session, method, endpoint, **kwargs)
                for _ in range(num_requests)
            ]
            
            # 并发执行
            start_time = time.time()
            results = await asyncio.gather(*tasks)
            total_duration = time.time() - start_time
            
            # 统计分析
            success_count = sum(1 for r in results if r["success"])
            durations = [r["duration_ms"] for r in results]
            
            # 计算百分位数
            sorted_durations = sorted(durations)
            p50 = sorted_durations[int(len(durations) * 0.50)]
            p95 = sorted_durations[int(len(durations) * 0.95)]
            p99 = sorted_durations[int(len(durations) * 0.99)]
            
            return {
                "total_requests": num_requests,
                "success_count": success_count,
                "failed_count": num_requests - success_count,
                "success_rate": success_count / num_requests * 100,
                "total_duration_s": total_duration,
                "throughput_rps": num_requests / total_duration,  # 吞吐量（请求/秒）
                "response_time": {
                    "min_ms": min(durations),
                    "max_ms": max(durations),
                    "avg_ms": mean(durations),
                    "median_ms": median(durations),
                    "p50_ms": p50,
                    "p95_ms": p95,
                    "p99_ms": p99
                },
                "errors": [r for r in results if not r["success"]]
            }

@pytest.mark.asyncio
async def test_concurrent_item_list():
    """测试零件列表并发访问（100并发用户）"""
    
    suite = PerformanceTestSuite("http://localhost:8000")
    
    result = await suite.concurrent_requests(
        method="GET",
        endpoint="/api/items",
        num_requests=100
    )
    
    # 断言性能指标
    assert result["success_rate"] > 99, f"成功率不足: {result['success_rate']}%"
    assert result["response_time"]["p95_ms"] < 200, f"P95响应时间过长: {result['response_time']['p95_ms']}ms"
    assert result["throughput_rps"] > 50, f"吞吐量过低: {result['throughput_rps']} RPS"
    
    # 打印性能报告
    print("\n=== 零件列表并发测试报告 ===")
    print(f"总请求数: {result['total_requests']}")
    print(f"成功数: {result['success_count']}")
    print(f"失败数: {result['failed_count']}")
    print(f"成功率: {result['success_rate']:.2f}%")
    print(f"吞吐量: {result['throughput_rps']:.2f} RPS")
    print(f"\n响应时间统计:")
    print(f"  最小值: {result['response_time']['min_ms']:.2f}ms")
    print(f"  平均值: {result['response_time']['avg_ms']:.2f}ms")
    print(f"  中位数: {result['response_time']['median_ms']:.2f}ms")
```

**Locust负载测试：**

Locust是基于Python的负载测试工具,支持编写复杂的用户行为场景,提供Web UI实时监控。

```python
# tests/locustfile.py - Locust测试脚本
from locust import HttpUser, task, between, events
from random import randint
import logging

class CampusTradingUser(HttpUser):
    """
    模拟汽配市场交易平台用户行为
    
    用户行为模式:
    - 浏览零件列表(高频)
    - 查看零件详情(中频)
    - 搜索零件(中频)
    - 发布零件(低频)
    - 创建交易(低频)
    
    等待时间: 1-3秒(模拟真实用户思考时间)
    """
    
    wait_time = between(1, 3)  # 请求间隔1-3秒
    
    def on_start(self):
        """
        用户启动时执行(登录获取token)
        
        每个虚拟用户启动时自动调用,模拟用户登录流程
        """
        response = self.client.post("/api/auth/login", json={
            "username": "testuser",
            "password": "Password123!"
        })
        
        if response.status_code == 200:
            self.token = response.json()["access_token"]
            self.headers = {"Authorization": f"Bearer {self.token}"}
            logging.info(f"User logged in successfully, token: {self.token[:20]}...")
        else:
            logging.error(f"Login failed: {response.status_code} - {response.text}")
            self.headers = {}
    
    @task(5)  # 权重5,执行频率最高
    def view_items(self):
        """浏览零件列表（最高频操作）"""
        page = randint(1, 10)
        self.client.get(
            f"/api/items?page={page}&page_size=20",
            headers=self.headers,
            name="/api/items (view list)"  # 自定义显示名称
        )
    
    @task(3)  # 权重3
    def view_item_detail(self):
        """查看零件详情"""
        item_id = randint(1, 100)
```

```python
# core/logging_config.py
import logging
from logging.handlers import RotatingFileHandler
import os

def setup_logging():
    """配置日志系统"""
    
    # 创建日志目录
    log_dir = "logs"
    os.makedirs(log_dir, exist_ok=True)
    
    # 配置格式
    formatter = logging.Formatter(
        '%(asctime)s - %(name)s - %(levelname)s - %(message)s'
    )
    
    # 文件处理器（自动轮转）
    file_handler = RotatingFileHandler(
        f"{log_dir}/app.log",
        maxBytes=10*1024*1024,  # 10MB
        backupCount=5
    )
    file_handler.setFormatter(formatter)
    
    # 控制台处理器
    console_handler = logging.StreamHandler()
    console_handler.setFormatter(formatter)
    
    # 配置根日志器
    root_logger = logging.getLogger()
    root_logger.setLevel(logging.INFO)
    root_logger.addHandler(file_handler)
    root_logger.addHandler(console_handler)
```
