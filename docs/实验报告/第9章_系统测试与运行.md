        "username": "duplicate",
        "email": "dup1@example.com",
        "password": "Password123!"
    })
    
    # 第二次注册（应失败）
    response = client.post("/api/auth/register", json={
        "username": "duplicate",
        "email": "dup2@example.com",
        "password": "Password123!"
    })
    
    assert response.status_code == 400
    assert "already exists" in response.json()["detail"].lower()

def test_version_conflict():
    """测试版本冲突检测"""
    
    item_id = 1
    
    # 在两个数据库同时更新同一记录
    with db_manager.session_scope("mysql") as session:
        item = session.get(Item, item_id)
        item.price = 100
        item.sync_version += 1
        session.commit()
    
    with db_manager.session_scope("postgres") as session:
        item = session.get(Item, item_id)
        item.price = 200
        item.sync_version += 1
        session.commit()
    
    # 触发同步，应检测到冲突
    time.sleep(3)
    
    # 查询冲突记录
    with db_manager.session_scope("mysql") as session:
        conflict = session.execute(
            select(ConflictRecord).where(
                ConflictRecord.table_name == "items",
                ConflictRecord.record_id == item_id
            )
        ).scalar_one_or_none()
        
        assert conflict is not None
        assert conflict.conflict_type == "version_conflict"

### 9.4.1 并发测试

```python
# tests/test_performance.py
import asyncio
import aiohttp
from concurrent.futures import ThreadPoolExecutor

async def concurrent_requests(url: str, num_requests: int):
    """并发请求测试"""
    
    async with aiohttp.ClientSession() as session:
        tasks = []
        for i in range(num_requests):
            task = session.get(url)
            tasks.append(task)
        
        responses = await asyncio.gather(*tasks)
        
        success = sum(1 for r in responses if r.status == 200)
        return {
            "total": num_requests,
            "success": success,
            "failed": num_requests - success,
            "success_rate": success / num_requests * 100
        }

def test_concurrent_item_list():
    """测试商品列表并发访问"""
    
    result = asyncio.run(
        concurrent_requests("http://localhost:8000/api/items", 100)
    )
    
    assert result["success_rate"] > 95  # 成功率应大于95%
```

### 9.5.3 日志管理

```python
# core/logging_config.py
import logging
from logging.handlers import RotatingFileHandler
import os

def setup_logging():
    """配置日志系统"""
    
    # 创建日志目录
    log_dir = "logs"
    os.makedirs(log_dir, exist_ok=True)
    
    # 配置格式
    formatter = logging.Formatter(
        '%(asctime)s - %(name)s - %(levelname)s - %(message)s'
    )
    
    # 文件处理器（自动轮转）
    file_handler = RotatingFileHandler(
        f"{log_dir}/app.log",
        maxBytes=10*1024*1024,  # 10MB
        backupCount=5
    )
    file_handler.setFormatter(formatter)
    
    # 控制台处理器
    console_handler = logging.StreamHandler()
    console_handler.setFormatter(formatter)
    
    # 配置根日志器
    root_logger = logging.getLogger()
    root_logger.setLevel(logging.INFO)
    root_logger.addHandler(file_handler)
    root_logger.addHandler(console_handler)
```
