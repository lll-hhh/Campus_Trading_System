```python
from fastapi import Depends, HTTPException, status
from fastapi.security import HTTPBearer, HTTPAuthorizationCredentials
from jose import JWTError, jwt
from typing import Optional

security = HTTPBearer()

# 获取当前用户
async def get_current_user(
    credentials: HTTPAuthorizationCredentials = Depends(security),
    db_name: str = "mysql"
) -> User:
    """从JWT令牌获取当前用户"""
    
    token = credentials.credentials
    
    try:
        # 解码JWT
        payload = jwt.decode(
            token, 
            settings.SECRET_KEY, 
            algorithms=[settings.ALGORITHM]
        )
        user_id: int = payload.get("sub")
        
        if user_id is None:
            raise HTTPException(
                status_code=status.HTTP_401_UNAUTHORIZED,
                detail="Invalid authentication credentials"
            )
            
    except JWTError:
        raise HTTPException(
            status_code=status.HTTP_401_UNAUTHORIZED,
            detail="Could not validate credentials"
        )
    
    # 从数据库查询用户
    with db_manager.session_scope(db_name) as session:
        user = session.get(User, user_id)
        
        if user is None:
            raise HTTPException(
                status_code=status.HTTP_404_NOT_FOUND,
                detail="User not found"
            )
        
        if not user.is_active:
            raise HTTPException(
                status_code=status.HTTP_403_FORBIDDEN,
                detail="Inactive user"
            )
    
    return user

# 权限检查
def require_permission(permission: str):
    """检查用户是否具有指定权限"""
    
    async def permission_checker(
        current_user: User = Depends(get_current_user)
    ):
        # 查询用户权限
        with db_manager.session_scope("mysql") as session:
            has_permission = session.execute(
                select(Permission)
                .join(RolePermission)
                .join(UserRole)
                .where(UserRole.user_id == current_user.id)
                .where(Permission.permission_code == permission)
            ).scalar_one_or_none()
            
            if not has_permission:
                raise HTTPException(
                    status_code=status.HTTP_403_FORBIDDEN,
                    detail=f"Permission denied: {permission}"
                )
        
        return current_user
    
    return permission_checker
```

```python
    """创建交易订单"""

                text("CALL sp_create_transaction(:item_id, :buyer_id, :quantity, @tx_id, @status, @message)"),
                {

                    "buyer_id": current_user.id,
                    "quantity": 1
                }
            )
            
            # 获取输出参数
            output = session.execute(
                text("SELECT @tx_id AS tx_id, @status AS status, @message AS message")
            ).fetchone()
            
            if output.status != "success":
                raise HTTPException(
                    status_code=status.HTTP_400_BAD_REQUEST,
                    detail=output.message
                )
            
            # 查询完整交易信息
            transaction = session.get(Transaction, output.tx_id)
            
            # 发布同步事件
            sync_event = SyncEvent(
                event_type="INSERT",
                table_name="transactions",
                source_db="mysql",
                record_id=transaction.id,
                data=transaction.to_dict()
            )
            await sync_engine.publish_event(sync_event)
            
            return transaction
            
    except Exception as e:
        logger.error(f"Failed to create transaction: {e}")
        raise HTTPException(
            status_code=status.HTTP_500_INTERNAL_SERVER_ERROR,
            detail="Failed to create transaction"
        )

@router.get("/", response_model=List[TransactionResponse])
async def list_transactions(
    status_filter: Optional[str] = None,
    pagination: PaginationParams = Depends(),
    current_user: User = Depends(get_current_user)
):
    """查询交易列表"""
    
    with db_manager.session_scope("mysql") as session:
        query = select(Transaction).where(
            or_(
                Transaction.buyer_id == current_user.id,
                Transaction.seller_id == current_user.id
            )
        )
        
        if status_filter:
            query = query.where(Transaction.status == status_filter)
        
        query = query.order_by(Transaction.created_at.desc())
        query = query.limit(pagination.page_size).offset(pagination.offset)
        
        transactions = session.execute(query).scalars().all()
        
    return transactions

@router.get("/{transaction_id}", response_model=TransactionDetailResponse)

        buyer = session.get(User, transaction.buyer_id)

    return {
        "transaction": transaction,
        "item": item,
        "buyer": buyer,
        "seller": seller

    """更新交易状态"""
    
    with db_manager.session_scope("mysql") as session:
        transaction = session.get(Transaction, transaction_id)

            raise HTTPException(status_code=404, detail="Transaction not found")
        
        # 状态流转验证
        valid_transitions = {
            "pending": ["paid", "cancelled"],
            "paid": ["shipped", "cancelled"],
            "shipped": ["delivered", "cancelled"],
            "delivered": ["completed", "refunded"],
        }
        
        if status_update.new_status not in valid_transitions.get(transaction.status, []):
            raise HTTPException(
                status_code=400,
                detail=f"Invalid status transition: {transaction.status} -> {status_update.new_status}"
            )
        
        # 更新状态
        transaction.status = status_update.new_status
        transaction.updated_at = datetime.utcnow()
        transaction.sync_version += 1
        
        session.commit()
        
        # 发送通知
        if status_update.new_status == "shipped":
            await send_notification(
                transaction.buyer_id,
                "订单已发货",
                f"您的订单 {transaction_id} 已发货"
            )
    
    return {"message": "Status updated successfully"}
```

```python
# core/database.py
from sqlalchemy import create_engine
from sqlalchemy.orm import sessionmaker, Session
from contextlib import contextmanager
from typing import Dict

class DatabaseManager:
    """多数据库管理器"""
    
    def __init__(self):
        self.engines: Dict[str, Engine] = {}
        self.session_factories: Dict[str, sessionmaker] = {}
    
    def initialize(self):
        """初始化所有数据库连接"""
        
        # MySQL主库
        self.engines["mysql"] = create_engine(
            settings.MYSQL_URL,
            pool_size=20,
            max_overflow=10,
            pool_pre_ping=True,
            echo=settings.DEBUG
        )
        
        # PostgreSQL
        self.engines["postgres"] = create_engine(
            settings.POSTGRES_URL,
            pool_size=10,
            max_overflow=5,
            pool_pre_ping=True
        )
        
        # MariaDB
        self.engines["mariadb"] = create_engine(
            settings.MARIADB_URL,
            pool_size=10,
            max_overflow=5,
            pool_pre_ping=True
        )
        
        # SQLite
        self.engines["sqlite"] = create_engine(
            settings.SQLITE_URL,
            connect_args={"check_same_thread": False}
        )
        
        # 创建会话工厂
        for db_name, engine in self.engines.items():
            self.session_factories[db_name] = sessionmaker(
                bind=engine,
                autocommit=False,
                autoflush=False
            )
        
        logger.info("✅ Database connections initialized")
    
    @contextmanager
    def session_scope(self, db_name: str = "mysql"):
        """提供事务性会话"""
        
        session = self.session_factories[db_name]()
        
        try:
            yield session
            session.commit()
        except Exception as e:
            session.rollback()
            logger.error(f"Database error in {db_name}: {e}")
            raise
        finally:
            session.close()
    
    def get_session(self, db_name: str = "mysql") -> Session:
        """获取会话（需要手动管理）"""
        return self.session_factories[db_name]()
    
    def close_all(self):
        """关闭所有数据库连接"""
        for db_name, engine in self.engines.items():
            engine.dispose()
            logger.info(f"Closed {db_name} connection")
```

```python
# apps/sync_service/engine.py
import redis
import json
from dataclasses import dataclass, asdict
from datetime import datetime
from typing import Dict, Any

@dataclass
class SyncEvent:
    """同步事件数据结构"""
    event_id: str
    event_type: str  # INSERT, UPDATE, DELETE
    table_name: str
    source_db: str
    record_id: int
    timestamp: datetime
    data: Dict[str, Any]
    
    def as_message(self) -> Dict:
        """转换为Redis消息格式"""
        return {
            "event_id": self.event_id,
            "event_type": self.event_type,
            "table_name": self.table_name,
            "source_db": self.source_db,
            "record_id": self.record_id,
            "timestamp": self.timestamp.isoformat(),
            "data": json.dumps(self.data)
        }

class SyncEngine:
    """同步引擎"""
    
    def __init__(self, redis_url: str):
        self.redis = redis.from_url(redis_url, decode_responses=True)
        self.stream_name = "campuswap:sync:events"
        self.consumer_group = "sync_workers"
    
    async def publish_event(self, event: SyncEvent):
        """发布同步事件到Redis Stream"""
        
        try:
            # 添加事件到Stream
            message_id = self.redis.xadd(
                self.stream_name,
                event.as_message(),
                maxlen=10000  # 限制Stream长度
            )
            
            logger.info(
                f"Published sync event: {event.event_type} "
                f"{event.table_name}:{event.record_id} -> {message_id}"
            )
            
            # 更新统计
            self.redis.hincrby("campuswap:metrics", "events_published", 1)
            
            return message_id
            
        except Exception as e:
            logger.error(f"Failed to publish sync event: {e}")
            raise
    
    async def create_consumer_group(self):
        """创建消费者组"""
        try:
            self.redis.xgroup_create(
                self.stream_name,
                self.consumer_group,
                id='0',
                mkstream=True
            )
            logger.info(f"Created consumer group: {self.consumer_group}")
        except redis.ResponseError as e:
            if "BUSYGROUP" not in str(e):
                raise
```

#### 同步Worker实现

```python
# apps/sync_service/worker.py
import asyncio
from typing import List

class SyncWorker:
    """同步Worker进程"""
    
    def __init__(self, worker_id: int, db_manager, sync_engine):
        self.worker_id = worker_id
        self.db_manager = db_manager
        self.sync_engine = sync_engine
        self.consumer_name = f"worker-{worker_id}"
        self.running = False
    
    async def start(self):
        """启动Worker"""
        self.running = True
        logger.info(f"Worker {self.worker_id} started")
        
        while self.running:
            try:
                # 从Stream读取事件
                events = self.sync_engine.redis.xreadgroup(
                    groupname=self.sync_engine.consumer_group,
                    consumername=self.consumer_name,
                    streams={self.sync_engine.stream_name: '>'},
                    count=10,
                    block=1000  # 阻塞1秒
                )
                
                if events:
                    for stream_name, messages in events:
                        for message_id, data in messages:
                            await self.process_event(message_id, data)
                
            except Exception as e:
                logger.error(f"Worker {self.worker_id} error: {e}")
                await asyncio.sleep(1)
    
    async def process_event(self, message_id: str, data: Dict):
        """处理同步事件"""
        
        try:
            # 解析事件数据
            event = SyncEvent(
                event_id=data["event_id"],
                event_type=data["event_type"],
                table_name=data["table_name"],
                source_db=data["source_db"],
                record_id=int(data["record_id"]),
                timestamp=datetime.fromisoformat(data["timestamp"]),
                data=json.loads(data["data"])
            )
            
            # 同步到其他数据库
            target_dbs = ["postgres", "mariadb", "sqlite"]
            if event.source_db in target_dbs:
                target_dbs.remove(event.source_db)
            
            for target_db in target_dbs:
                await self.sync_to_db(event, target_db)
            
            # 确认消息
            self.sync_engine.redis.xack(
                self.sync_engine.stream_name,
                self.sync_engine.consumer_group,
                message_id
            )
            
            # 更新统计
            self.sync_engine.redis.hincrby("campuswap:metrics", "events_consumed", 1)
            
            logger.debug(f"Processed event {event.event_id}")
            
        except Exception as e:
            logger.error(f"Failed to process event {message_id}: {e}")
    
    async def sync_to_db(self, event: SyncEvent, target_db: str):
        """同步数据到目标数据库"""
        
        try:
            with self.db_manager.session_scope(target_db) as session:
                if event.event_type == "INSERT":
                    # 插入记录
                    model_class = get_model_class(event.table_name)
                    record = model_class(**event.data)
                    session.add(record)
                    
                elif event.event_type == "UPDATE":
                    # 更新记录
                    model_class = get_model_class(event.table_name)
                    record = session.get(model_class, event.record_id)
                    
                    if record:
                        for key, value in event.data.items():
                            setattr(record, key, value)
                    else:
                        # 记录不存在，插入新记录
                        record = model_class(**event.data)
                        session.add(record)
                    
                elif event.event_type == "DELETE":
                    # 删除记录
                    model_class = get_model_class(event.table_name)
                    record = session.get(model_class, event.record_id)
                    if record:
                        session.delete(record)
                
                session.commit()
                
                logger.debug(
                    f"Synced {event.event_type} {event.table_name}:{event.record_id} "
                    f"from {event.source_db} to {target_db}"
                )
                
        except Exception as e:
            logger.error(
                f"Failed to sync to {target_db}: {e}",
                extra={
                    "event": event.event_id,
                    "table": event.table_name,
                    "record_id": event.record_id
                }
            )
            
            # 记录冲突
            await self.record_conflict(event, target_db, str(e))
    
    async def record_conflict(self, event: SyncEvent, target_db: str, error: str):
        """记录同步冲突"""
        
        with self.db_manager.session_scope("mysql") as session:
            conflict = ConflictRecord(
                table_name=event.table_name,
                record_id=event.record_id,
                source_db=event.source_db,
                target_db=target_db,
                conflict_type="sync_error",
                source_data=event.data,
                target_data={},
                error_message=error,
                status="pending",
                created_at=datetime.utcnow()
            )
            session.add(conflict)
            session.commit()

# 启动多个Worker
async def start_sync_workers(num_workers: int = 4):
    """启动多个同步Worker"""
    
    workers = []
    for i in range(num_workers):
        worker = SyncWorker(i, db_manager, sync_engine)
        workers.append(worker)
        asyncio.create_task(worker.start())
    
    logger.info(f"Started {num_workers} sync workers")
    return workers
```

### 6.2.2 Pinia状态管理

```typescript
// src/stores/item.ts
import { defineStore } from 'pinia'
import { ref, computed } from 'vue'
import type { Item, ItemFilter } from '@/types'
import { api } from '@/lib/api'

export const useItemStore = defineStore('item', () => {
  // 状态
  const items = ref<Item[]>([])
  const currentItem = ref<Item | null>(null)
  const loading = ref(false)
  const error = ref<string | null>(null)
  
  // 分页
  const page = ref(1)
  const pageSize = ref(20)
  const total = ref(0)
  
  // 过滤条件
  const filter = ref<ItemFilter>({
    status: 'available',
    categoryId: null,
    minPrice: null,
    maxPrice: null,
    keyword: ''
  })
  
  // 计算属性
  const hasMore = computed(() => {
    return items.value.length < total.value
  })
  
  const availableItems = computed(() => {
    return items.value.filter(item => item.status === 'available')
  })
  
  // 操作
  async function fetchItems(resetPage = false) {
    if (resetPage) {
      page.value = 1
      items.value = []
    }
    
    loading.value = true
    error.value = null
    
    try {
      const response = await api.get('/items', {
        params: {
          page: page.value,
          page_size: pageSize.value,
          ...filter.value
        }
      })
      
      if (resetPage) {
        items.value = response.data.items
      } else {
        items.value.push(...response.data.items)
      }
      
      total.value = response.data.total
      page.value++
      
    } catch (err: any) {
      error.value = err.message || 'Failed to fetch items'
      throw err
    } finally {
      loading.value = false
    }
  }
  
  async function fetchItemDetail(itemId: number) {
    loading.value = true
    error.value = null
    
    try {
      const response = await api.get(`/items/${itemId}`)
      currentItem.value = response.data
      
      // 增加浏览次数
      await api.post(`/items/${itemId}/view`)
      
    } catch (err: any) {
      error.value = err.message || 'Failed to fetch item detail'
      throw err
    } finally {
      loading.value = false
    }
  }
  
  async function createItem(itemData: Partial<Item>) {
    loading.value = true
    error.value = null
    
    try {
      const response = await api.post('/items', itemData)
      items.value.unshift(response.data)
      return response.data
    } catch (err: any) {
      error.value = err.message || 'Failed to create item'
      throw err
    } finally {
      loading.value = false
    }
  }
  
  async function updateItem(itemId: number, itemData: Partial<Item>) {
    loading.value = true
    error.value = null
    
    try {
      const response = await api.patch(`/items/${itemId}`, itemData)
      
      // 更新本地状态
      const index = items.value.findIndex(item => item.id === itemId)
      if (index !== -1) {
        items.value[index] = response.data
      }
      
      if (currentItem.value?.id === itemId) {
        currentItem.value = response.data
      }
      
      return response.data
    } catch (err: any) {
      error.value = err.message || 'Failed to update item'
      throw err
    } finally {
      loading.value = false
    }
  }
  
  async function favoriteItem(itemId: number) {
    await api.post(`/items/${itemId}/favorite`)
    
    // 更新收藏数
    const item = items.value.find(i => i.id === itemId)
    if (item) {
      item.favoriteCount++
    }
  }
  
  async function unfavoriteItem(itemId: number) {
    await api.delete(`/items/${itemId}/favorite`)
    
    // 更新收藏数
    const item = items.value.find(i => i.id === itemId)
    if (item) {
      item.favoriteCount--
    }
  }
  
  function updateFilter(newFilter: Partial<ItemFilter>) {
    filter.value = { ...filter.value, ...newFilter }
  }
  
  function resetFilter() {
    filter.value = {
      status: 'available',
      categoryId: null,
      minPrice: null,
      maxPrice: null,
      keyword: ''
    }
  }
  
  return {
    // 状态
    items,
    currentItem,
    loading,
    error,
    page,
    pageSize,
    total,
    filter,
    
    // 计算属性
    hasMore,
    availableItems,
    
    // 操作
    fetchItems,
    fetchItemDetail,
    createItem,
    updateItem,
    favoriteItem,
    unfavoriteItem,
    updateFilter,
    resetFilter
  }
})
```

### 6.2.3 WebSocket实时通信

```typescript
// src/lib/websocket.ts
import { ref, onUnmounted } from 'vue'
import { useAuthStore } from '@/stores/auth'

export class WebSocketClient {
  private ws: WebSocket | null = null
  private reconnectTimer: number | null = null
  private heartbeatTimer: number | null = null
  private messageHandlers: Map<string, Function[]> = new Map()
  
  public connected = ref(false)
  public connecting = ref(false)
  
  constructor(private url: string) {}
  
  connect() {
    if (this.ws?.readyState === WebSocket.OPEN) {
      return
    }
    
    this.connecting.value = true
    
    const authStore = useAuthStore()
    const token = authStore.token
    
    // 连接WebSocket
    this.ws = new WebSocket(`${this.url}?token=${token}`)
    
    this.ws.onopen = () => {
      console.log('WebSocket connected')
      this.connected.value = true
      this.connecting.value = false
      this.startHeartbeat()
    }
    
    this.ws.onmessage = (event) => {
      try {
        const message = JSON.parse(event.data)
        this.handleMessage(message)
      } catch (error) {
        console.error('Failed to parse WebSocket message:', error)
      }
    }
    
    this.ws.onerror = (error) => {
      console.error('WebSocket error:', error)
    }
    
    this.ws.onclose = () => {
      console.log('WebSocket disconnected')
      this.connected.value = false
      this.connecting.value = false
      this.stopHeartbeat()
      this.scheduleReconnect()
    }
  }
  
  disconnect() {
    if (this.reconnectTimer) {
      clearTimeout(this.reconnectTimer)
      this.reconnectTimer = null
    }
    
    if (this.ws) {
      this.ws.close()
      this.ws = null
    }
  }
  
  send(type: string, data: any) {
    if (this.ws?.readyState === WebSocket.OPEN) {
      this.ws.send(JSON.stringify({ type, data }))
    } else {
      console.warn('WebSocket is not connected')
    }
  }
  
  on(type: string, handler: Function) {
    if (!this.messageHandlers.has(type)) {
      this.messageHandlers.set(type, [])
    }
    this.messageHandlers.get(type)!.push(handler)
  }
  
  off(type: string, handler: Function) {
    const handlers = this.messageHandlers.get(type)
    if (handlers) {
      const index = handlers.indexOf(handler)
      if (index !== -1) {
        handlers.splice(index, 1)
      }
    }
  }
  
  private handleMessage(message: any) {
    const handlers = this.messageHandlers.get(message.type)
    if (handlers) {
      handlers.forEach(handler => handler(message.data))
    }
  }
  
  private startHeartbeat() {
    this.heartbeatTimer = window.setInterval(() => {
      this.send('ping', {})
    }, 30000) // 每30秒发送心跳
  }
  
  private stopHeartbeat() {
    if (this.heartbeatTimer) {
      clearInterval(this.heartbeatTimer)
      this.heartbeatTimer = null
    }
  }
  
  private scheduleReconnect() {
    if (this.reconnectTimer) return
    
    this.reconnectTimer = window.setTimeout(() => {
      console.log('Attempting to reconnect WebSocket...')
      this.reconnectTimer = null
      this.connect()
    }, 5000) // 5秒后重连
  }
}

// 全局WebSocket实例
export const wsClient = new WebSocketClient(import.meta.env.VITE_WS_URL)

// 在组件中使用
export function useWebSocket() {
  onUnmounted(() => {
    // 组件卸载时不断开连接，保持全局连接
  })
  
  return wsClient
}
```
