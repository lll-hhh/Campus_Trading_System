```python
from datetime import datetime, timedelta
from typing import Dict, List

class SyncStatistics:
    """同步统计服务"""
    
    def get_sync_overview(self, days: int = 7) -> Dict:
        """获取同步概览统计"""
        
        with db_manager.session_scope("mysql") as session:
            end_date = datetime.utcnow()
            start_date = end_date - timedelta(days=days)
            
            # 总体统计
            total_stats = session.execute(
                select(
                    func.count(SyncLog.id).label("total_syncs"),
                    func.sum(case((SyncLog.status == "success", 1), else_=0)).label("success_count"),
                    func.sum(case((SyncLog.status == "failed", 1), else_=0)).label("failed_count"),
                    func.avg(SyncLog.duration_ms).label("avg_duration"),
                    func.sum(SyncLog.records_synced).label("total_records")
                ).where(SyncLog.started_at >= start_date)
            ).fetchone()
            
            # 按数据库统计
            db_stats = session.execute(
                select(
                    SyncLog.target_db,
                    func.count(SyncLog.id).label("sync_count"),
                    func.avg(SyncLog.duration_ms).label("avg_duration")
                )
                .where(SyncLog.started_at >= start_date)
                .group_by(SyncLog.target_db)
            ).fetchall()
            
            # 按表统计
            table_stats = session.execute(
                select(
                    SyncLog.table_name,
                    func.count(SyncLog.id).label("sync_count"),
                    func.sum(SyncLog.records_synced).label("records_synced")
                )
                .where(SyncLog.started_at >= start_date)
                .group_by(SyncLog.table_name)
                .order_by(desc("sync_count"))
                .limit(10)
            ).fetchall()
            
            return {
                "period": f"Last {days} days",
                "overview": {
                    "total_syncs": total_stats.total_syncs or 0,
                    "success_count": total_stats.success_count or 0,
                    "failed_count": total_stats.failed_count or 0,
                    "success_rate": (
                        total_stats.success_count / total_stats.total_syncs * 100
                        if total_stats.total_syncs else 100
                    ),
                    "avg_duration_ms": round(total_stats.avg_duration or 0, 2),
                    "total_records": total_stats.total_records or 0
                },
                "by_database": [
                    {
                        "database": row.target_db,
                        "sync_count": row.sync_count,
                        "avg_duration": round(row.avg_duration, 2)
                    }
                    for row in db_stats
                ],
                "by_table": [
                    {
                        "table": row.table_name,
                        "sync_count": row.sync_count,
                        "records_synced": row.records_synced
                    }
                    for row in table_stats
                ]
            }
```

```python
            select(

                func.count(SyncLog.id).label("total"),
                func.sum(case((SyncLog.status == "success", 1), else_=0)).label("success"),
                func.avg(SyncLog.duration_ms).label("avg_duration"),
                func.max(SyncLog.duration_ms).label("max_duration")
            )
            .where(SyncLog.started_at >= start_time)
            .group_by("hour")
            .order_by("hour")
        ).fetchall()
        
        return {
            "period": f"Last {hours} hours",
            "data_points": [
                {
                    "time": row.hour,
                    "total": row.total,
                    "success": row.success,
                    "success_rate": (row.success / row.total * 100) if row.total else 100,
                    "avg_duration": round(row.avg_duration, 2),
                    "max_duration": round(row.max_duration, 2)
                }
                for row in hourly_stats
            ]
        }
```

```python
class BusinessStatistics:
    """业务统计服务"""
    
    def get_transaction_statistics(self, days: int = 30) -> Dict:
        """获取交易统计"""
        
        with db_manager.session_scope("mysql") as session:
            end_date = datetime.utcnow()
            start_date = end_date - timedelta(days=days)
            
            # 交易概览
            overview = session.execute(
                select(
                    func.count(Transaction.id).label("total_transactions"),
                    func.sum(
                        case((Transaction.status == "completed", 1), else_=0)
                    ).label("completed"),
                    func.sum(
                        case((Transaction.status == "completed", Transaction.total_price), else_=0)
                    ).label("total_amount"),
                    func.avg(
                        case((Transaction.status == "completed", Transaction.total_price), else_=None)
                    ).label("avg_amount")
                ).where(Transaction.created_at >= start_date)
            ).fetchone()
            
            # 每日交易趋势
            daily_trends = session.execute(
                select(
                    func.date(Transaction.created_at).label("date"),
                    func.count(Transaction.id).label("count"),
                    func.sum(
                        case((Transaction.status == "completed", Transaction.total_price), else_=0)
                    ).label("amount")
                )
                .where(Transaction.created_at >= start_date)
                .group_by("date")
                .order_by("date")
            ).fetchall()
            
            # 分类交易统计
            category_stats = session.execute(
                select(
                    Category.name,
                    func.count(Transaction.id).label("transaction_count"),
                    func.sum(Transaction.total_price).label("total_amount")
                )
                .join(Item, Transaction.item_id == Item.id)
                .join(Category, Item.category_id == Category.id)
                .where(
                    Transaction.created_at >= start_date,
                    Transaction.status == "completed"
                )
                .group_by(Category.name)
                .order_by(desc("total_amount"))
                .limit(10)
            ).fetchall()
            
            return {
                "overview": {
                    "total_transactions": overview.total_transactions or 0,
                    "completed_transactions": overview.completed or 0,
                    "completion_rate": (
                        overview.completed / overview.total_transactions * 100
                        if overview.total_transactions else 0
                    ),
                    "total_amount": float(overview.total_amount or 0),
                    "avg_amount": float(overview.avg_amount or 0)
                },
                "daily_trends": [
                    {
                        "date": row.date.isoformat(),
                        "count": row.count,
                        "amount": float(row.amount)
                    }
                    for row in daily_trends
                ],
                "by_category": [
                    {
                        "category": row.name,
                        "transaction_count": row.transaction_count,
                        "total_amount": float(row.total_amount)
                    }
                    for row in category_stats
                ]
            }
```

```python
def get_user_activity_statistics(self, days: int = 30) -> Dict:
    """获取用户活跃度统计"""
    
    with db_manager.session_scope("mysql") as session:
        end_date = datetime.utcnow()
        start_date = end_date - timedelta(days=days)
        
        # DAU（日活跃用户）
        dau_data = session.execute(
            select(
                func.date(created_at).label("date"),
                func.count(func.distinct(user_id)).label("active_users")
            )
            .select_from(
                select(User.id.label("user_id"), Item.created_at)
                .select_from(Item)
                .union_all(
                    select(Transaction.buyer_id, Transaction.created_at)
                    .select_from(Transaction)
                )
                .union_all(
                    select(Message.sender_id, Message.created_at)
                    .select_from(Message)
                )
                .subquery()
            )
            .where(text("created_at >= :start_date"))
            .group_by("date")
            .order_by("date"),
            {"start_date": start_date}
        ).fetchall()
        
        # 用户留存率
        cohort_data = session.execute(
            text("""
                WITH user_cohorts AS (
                    SELECT 
                        id,
                        DATE_FORMAT(created_at, '%Y-%m') AS cohort_month
                    FROM users
                    WHERE created_at >= :start_date
                )
                SELECT 
                    cohort_month,
                    COUNT(DISTINCT id) AS cohort_size
                FROM user_cohorts
                GROUP BY cohort_month
                ORDER BY cohort_month
            """),
            {"start_date": start_date}
        ).fetchall()
        
        return {
            "dau": [
                {
                    "date": row.date.isoformat(),
                    "active_users": row.active_users
                }
                for row in dau_data
            ],
            "retention": [
                {
                    "cohort": row.cohort_month,
                    "size": row.cohort_size
                }
                for row in cohort_data
            ]
        }
```

```vue
<script setup lang="ts">
import { ref, onMounted, watch } from 'vue'
import * as echarts from 'echarts'
import { api } from '@/lib/api'

const chartRef = ref<HTMLDivElement>()
let chart: echarts.ECharts | null = null

const props = defineProps<{
  days?: number
}>()

onMounted(async () => {
  // 初始化图表
  chart = echarts.init(chartRef.value!)
  
  // 加载数据
  await loadChartData()
})

const loadChartData = async () => {
  try {
    const response = await api.get('/monitoring/sync/performance', {
      params: { days: props.days || 7 }
    })
    
    const data = response.data
    
    // 配置图表
    const option = {
      title: {
        text: '同步性能趋势',
        left: 'center'
      },
      tooltip: {
        trigger: 'axis'
      },
      legend: {
        data: ['同步次数', '成功率', '平均耗时'],
        top: 30
      },
      grid: {
        left: '3%',
        right: '4%',
        bottom: '3%',
        containLabel: true
      },
      xAxis: {
        type: 'category',
        boundaryGap: false,
        data: data.data_points.map(d => d.time)
      },
      yAxis: [
        {
          type: 'value',
          name: '次数',
          position: 'left'
        },
        {
          type: 'value',
          name: '成功率(%)',
          position: 'right',
          max: 100
        },
        {
          type: 'value',
          name: '耗时(ms)',
          position: 'right',
          offset: 60
        }
      ],
      series: [
        {
          name: '同步次数',
          type: 'line',
          data: data.data_points.map(d => d.total),
          smooth: true,
          yAxisIndex: 0
        },
        {
          name: '成功率',
          type: 'line',
          data: data.data_points.map(d => d.success_rate),
          smooth: true,
          yAxisIndex: 1,
          itemStyle: { color: '#52c41a' }

        {
          name: '平均耗时',

          smooth: true,
          yAxisIndex: 2,
          itemStyle: { color: '#faad14' }

      ]
    }
    
    chart?.setOption(option)
  } catch (error) {
    console.error('Failed to load chart data:', error)
  }
}

// 监听props变化

})
```

### 8.4.1 Excel报表导出

```python
from openpyxl import Workbook
from openpyxl.styles import Font, Alignment, PatternFill
from io import BytesIO

@router.get("/reports/sync/export")
async def export_sync_report(
    start_date: date,
    end_date: date,
    current_user: User = Depends(get_current_user)
):
    """导出同步报表（Excel格式）"""
    
    # 查询数据
    with db_manager.session_scope("mysql") as session:
        sync_logs = session.execute(
            select(SyncLog)
            .where(
                SyncLog.started_at >= start_date,
                SyncLog.started_at <= end_date
            )
            .order_by(SyncLog.started_at.desc())
        ).scalars().all()
    
    # 创建工作簿
    wb = Workbook()
    ws = wb.active
    ws.title = "同步日志"
    
    # 设置表头
    headers = ["ID", "表名", "源库", "目标库", "状态", "记录数", "耗时(ms)", "开始时间"]
    ws.append(headers)
    
    # 设置表头样式
    header_fill = PatternFill(start_color="366092", end_color="366092", fill_type="solid")
    header_font = Font(color="FFFFFF", bold=True)
    
    for cell in ws[1]:
        cell.fill = header_fill
        cell.font = header_font
        cell.alignment = Alignment(horizontal="center")
    
    # 填充数据
    for log in sync_logs:
        ws.append([
            log.id,
            log.table_name,
            log.source_db,
            log.target_db,
            log.status,
            log.records_synced,
            log.duration_ms,
            log.started_at.strftime("%Y-%m-%d %H:%M:%S")
        ])
    
    # 调整列宽
    ws.column_dimensions['A'].width = 8
    ws.column_dimensions['B'].width = 20
    ws.column_dimensions['C'].width = 12
    ws.column_dimensions['D'].width = 12
    ws.column_dimensions['E'].width = 10
    ws.column_dimensions['F'].width = 10
    ws.column_dimensions['G'].width = 12
    ws.column_dimensions['H'].width = 20
    
    # 保存到内存
    output = BytesIO()
    wb.save(output)
    output.seek(0)
    
    # 返回文件
    return StreamingResponse(
        output,
        media_type="application/vnd.openxmlformats-officedocument.spreadsheetml.sheet",
        headers={
            "Content-Disposition": f"attachment; filename=sync_report_{start_date}_{end_date}.xlsx"
        }
    )
```
