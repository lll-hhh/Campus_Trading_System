# 第4章 数据库编程

## 4.1 存储过程设计与实现

存储过程（Stored Procedure）是预编译的SQL语句集合，存储在数据库中，可以被重复调用。本系统设计了多个存储过程来实现复杂的业务逻辑。

### 4.1.1 用户注册存储过程

**功能说明：** 处理用户注册，包括密码加密、初始数据设置、审计日志记录。

```sql
-- MySQL实现
DELIMITER $$

CREATE PROCEDURE sp_register_user(
    IN p_username VARCHAR(50),
    IN p_email VARCHAR(100),
    IN p_password VARCHAR(255),
    IN p_phone VARCHAR(20),
    OUT p_user_id INT,
    OUT p_status VARCHAR(20),
    OUT p_message VARCHAR(255)
)
BEGIN
    DECLARE v_exists INT DEFAULT 0;
    DECLARE v_hashed_password VARCHAR(255);
    DECLARE EXIT HANDLER FOR SQLEXCEPTION
    BEGIN
        ROLLBACK;
        SET p_status = 'error';
        SET p_message = '注册失败，请稍后重试';
    END;
    
    START TRANSACTION;
    
    -- 1. 检查用户名是否已存在
    SELECT COUNT(*) INTO v_exists 
    FROM users 
    WHERE username = p_username OR email = p_email;
    
    IF v_exists > 0 THEN
        SET p_status = 'error';
        SET p_message = '用户名或邮箱已存在';
        ROLLBACK;
    ELSE
        -- 2. 插入用户记录
        INSERT INTO users (
            username, email, password_hash, phone,
            is_active, is_verified, total_items_sold,
            total_items_purchased, rating, created_at,
            updated_at, sync_version
        ) VALUES (
            p_username, p_email, p_password, p_phone,
            TRUE, FALSE, 0, 0, 5.0, NOW(), NOW(), 1
        );
        
        SET p_user_id = LAST_INSERT_ID();
        
        -- 3. 创建用户档案
        INSERT INTO user_profiles (
            user_id, nickname, avatar_url, bio,
            school, major, grade, created_at, updated_at
        ) VALUES (
            p_user_id, p_username, '/static/images/default-avatar.png',
            '这个人很懒，什么都没写', '', '', '', NOW(), NOW()
        );
        
        -- 4. 分配默认角色（普通用户）
        INSERT INTO user_roles (user_id, role_id, created_at)
        SELECT p_user_id, id, NOW()
        FROM roles
        WHERE role_name = 'user'
        LIMIT 1;
        
        -- 5. 记录审计日志
        INSERT INTO audit_logs (
            user_id, action, table_name, record_id,
            details, ip_address, created_at
        ) VALUES (
            p_user_id, 'CREATE', 'users', p_user_id,
            CONCAT('New user registered: ', p_username),
            '127.0.0.1', NOW()
        );
        
        SET p_status = 'success';
        SET p_message = '注册成功';
        
        COMMIT;
    END IF;
END$$

DELIMITER ;
```

```sql

    @status,                       -- 输出：状态
    @message                       -- 输出：消息

-- 查看结果
```

### 4.1.2 零件发布存储过程

**功能说明：** 处理零件发布，包括数据验证、图片处理、分类关联。

```sql
DELIMITER $$

CREATE PROCEDURE sp_publish_item(
    IN p_seller_id INT,
    IN p_title VARCHAR(200),
    IN p_description TEXT,
    IN p_price DECIMAL(10,2),
    IN p_category_id INT,
    IN p_condition VARCHAR(20),
    IN p_images JSON,
    IN p_location VARCHAR(100),
    OUT p_item_id INT,
    OUT p_status VARCHAR(20),
    OUT p_message VARCHAR(255)
)
BEGIN
    DECLARE v_seller_active BOOLEAN DEFAULT FALSE;
    DECLARE v_category_exists INT DEFAULT 0;
    DECLARE v_image_path VARCHAR(255);
    DECLARE v_image_index INT DEFAULT 0;
    DECLARE EXIT HANDLER FOR SQLEXCEPTION
    BEGIN
        ROLLBACK;
        SET p_status = 'error';
        SET p_message = '发布失败';
    END;
    
    START TRANSACTION;
    
    -- 1. 验证供应商状态
    SELECT is_active INTO v_seller_active
    FROM users
    WHERE id = p_seller_id;
    
    IF NOT v_seller_active THEN
        SET p_status = 'error';
        SET p_message = '账户未激活，无法发布零件';
        ROLLBACK;
    ELSE
        -- 2. 验证分类存在
        SELECT COUNT(*) INTO v_category_exists
        FROM categories
        WHERE id = p_category_id;
        
        IF v_category_exists = 0 THEN
            SET p_status = 'error';
            SET p_message = '零件分类不存在';
            ROLLBACK;
        ELSE
            -- 3. 插入零件记录
            INSERT INTO items (
                seller_id, title, description, price,
                category_id, item_condition, status,
                view_count, favorite_count, location,
                created_at, updated_at, sync_version
            ) VALUES (
                p_seller_id, p_title, p_description, p_price,
                p_category_id, p_condition, 'available',
                0, 0, p_location, NOW(), NOW(), 1
            );
            
            SET p_item_id = LAST_INSERT_ID();
            
            -- 4. 处理零件图片
            IF p_images IS NOT NULL AND JSON_LENGTH(p_images) > 0 THEN
                WHILE v_image_index < JSON_LENGTH(p_images) DO
                    SET v_image_path = JSON_UNQUOTE(JSON_EXTRACT(p_images, CONCAT('$[', v_image_index, ']')));
                    
                    INSERT INTO item_images (
                        item_id, image_url, display_order, created_at
                    ) VALUES (
                        p_item_id, v_image_path, v_image_index, NOW()
                    );
                    
                    SET v_image_index = v_image_index + 1;
                END WHILE;
            END IF;
            
            -- 5. 更新用户统计
            UPDATE users
            SET total_items_sold = total_items_sold + 1,
                updated_at = NOW(),
                sync_version = sync_version + 1
            WHERE id = p_seller_id;
            
            -- 6. 记录审计日志
            INSERT INTO audit_logs (
                user_id, action, table_name, record_id,
                details, created_at
            ) VALUES (
                p_seller_id, 'CREATE', 'items', p_item_id,
                CONCAT('Published item: ', p_title), NOW()
            );
            
            SET p_status = 'success';
            SET p_message = '零件发布成功';
            
            COMMIT;
        END IF;
    END IF;
END$$

DELIMITER ;
```

### 4.1.3 交易创建存储过程

**功能说明：** 处理交易创建，包括库存检查、状态更新、通知发送。

```sql
DELIMITER $$

CREATE PROCEDURE sp_create_transaction(
    IN p_item_id INT,
    IN p_buyer_id INT,
    IN p_quantity INT,
    OUT p_transaction_id INT,
    OUT p_status VARCHAR(20),
    OUT p_message VARCHAR(255)
)
BEGIN
    DECLARE v_seller_id INT;
    DECLARE v_item_price DECIMAL(10,2);
    DECLARE v_item_status VARCHAR(20);
    DECLARE v_buyer_active BOOLEAN;
    DECLARE EXIT HANDLER FOR SQLEXCEPTION
    BEGIN
        ROLLBACK;
        SET p_status = 'error';
        SET p_message = '交易创建失败';
    END;
    
    START TRANSACTION;
    
    -- 1. 获取零件信息并锁定记录
    SELECT seller_id, price, status
    INTO v_seller_id, v_item_price, v_item_status
    FROM items
    WHERE id = p_item_id
    FOR UPDATE;
    
    -- 2. 验证零件状态
    IF v_item_status != 'available' THEN
        SET p_status = 'error';
        SET p_message = '零件已下架或已售出';
        ROLLBACK;
    ELSE
        -- 3. 验证采购商不能是供应商
        IF p_buyer_id = v_seller_id THEN
            SET p_status = 'error';
            SET p_message = '不能购买自己的零件';
            ROLLBACK;
        ELSE
            -- 4. 验证采购商状态
            SELECT is_active INTO v_buyer_active
            FROM users
            WHERE id = p_buyer_id;
            
            IF NOT v_buyer_active THEN
                SET p_status = 'error';
                SET p_message = '采购商账户未激活';
                ROLLBACK;
            ELSE
                -- 5. 创建交易记录
                INSERT INTO transactions (
                    item_id, buyer_id, seller_id,
                    item_price, quantity, total_price,
                    status, created_at, updated_at, sync_version
                ) VALUES (
                    p_item_id, p_buyer_id, v_seller_id,
                    v_item_price, p_quantity, v_item_price * p_quantity,
                    'pending', NOW(), NOW(), 1
                );
                
                SET p_transaction_id = LAST_INSERT_ID();
                
                -- 6. 更新零件状态
                UPDATE items
                SET status = 'reserved',
                    updated_at = NOW(),
                    sync_version = sync_version + 1
                WHERE id = p_item_id;
                
                -- 7. 创建通知
                INSERT INTO notifications (
                    user_id, type, title, content,
                    related_type, related_id, is_read, created_at
                ) VALUES (
                    v_seller_id, 'transaction', '新的交易订单',
                    CONCAT('您的零件收到了新订单，订单号：', p_transaction_id),
                    'transaction', p_transaction_id, FALSE, NOW()
                );
                
                -- 8. 记录审计日志
                INSERT INTO audit_logs (
                    user_id, action, table_name, record_id,
                    details, created_at
                ) VALUES (
                    p_buyer_id, 'CREATE', 'transactions', p_transaction_id,
                    CONCAT('Created transaction for item ', p_item_id), NOW()
                );
                
                SET p_status = 'success';
                SET p_message = '订单创建成功';
                
                COMMIT;
            END IF;
        END IF;
    END IF;
END$$

DELIMITER ;
```

### 4.1.4 数据统计存储过程

**功能说明：** 生成每日统计报表，汇总各类业务数据。

```sql
DELIMITER $$

CREATE PROCEDURE sp_generate_daily_stats(
    IN p_stat_date DATE,
    OUT p_status VARCHAR(20)
)
BEGIN
    DECLARE v_new_users INT DEFAULT 0;
    DECLARE v_new_items INT DEFAULT 0;
    DECLARE v_new_transactions INT DEFAULT 0;
    DECLARE v_completed_transactions INT DEFAULT 0;
    DECLARE v_total_amount DECIMAL(12,2) DEFAULT 0;
    DECLARE v_new_messages INT DEFAULT 0;
    DECLARE v_sync_tasks INT DEFAULT 0;
    DECLARE v_sync_conflicts INT DEFAULT 0;
    
    -- 1. 统计新增用户
    SELECT COUNT(*) INTO v_new_users
    FROM users
    WHERE DATE(created_at) = p_stat_date;
    
    -- 2. 统计新发布零件
    SELECT COUNT(*) INTO v_new_items
    FROM items
    WHERE DATE(created_at) = p_stat_date;
    
    -- 3. 统计新建交易
    SELECT COUNT(*) INTO v_new_transactions
    FROM transactions
    WHERE DATE(created_at) = p_stat_date;
    
    -- 4. 统计完成交易和金额
    SELECT COUNT(*), COALESCE(SUM(total_price), 0)
    INTO v_completed_transactions, v_total_amount
    FROM transactions
    WHERE DATE(completed_at) = p_stat_date
    AND status = 'completed';
    
    -- 5. 统计新消息
    SELECT COUNT(*) INTO v_new_messages
    FROM messages
    WHERE DATE(created_at) = p_stat_date;
    
    -- 6. 统计同步任务
    SELECT COUNT(*) INTO v_sync_tasks
    FROM sync_logs
    WHERE DATE(started_at) = p_stat_date;
    
    -- 7. 统计同步冲突
    SELECT COUNT(*) INTO v_sync_conflicts
    FROM conflict_records
    WHERE DATE(created_at) = p_stat_date;
    
    -- 8. 插入或更新统计记录
    INSERT INTO daily_stats (
        stat_date, new_users, new_items,
        new_transactions, completed_transactions,
        total_transaction_amount, new_messages,
        sync_tasks, sync_conflicts, created_at
    ) VALUES (
        p_stat_date, v_new_users, v_new_items,
        v_new_transactions, v_completed_transactions,
        v_total_amount, v_new_messages,
        v_sync_tasks, v_sync_conflicts, NOW()
    )
    ON DUPLICATE KEY UPDATE
        new_users = v_new_users,
        new_items = v_new_items,
        new_transactions = v_new_transactions,
        completed_transactions = v_completed_transactions,
        total_transaction_amount = v_total_amount,
        new_messages = v_new_messages,
        sync_tasks = v_sync_tasks,
        sync_conflicts = v_sync_conflicts,
        updated_at = NOW();
    
    SET p_status = 'success';
END$$

DELIMITER ;
```

### 4.1.5 批量数据导入存储过程

**功能说明：** 批量导入零件数据，用于数据迁移或批量上架。

```sql
DELIMITER $$

CREATE PROCEDURE sp_batch_import_items(
    IN p_seller_id INT,
    IN p_items_json JSON,
    OUT p_success_count INT,
    OUT p_failed_count INT,
    OUT p_status VARCHAR(20)
)
BEGIN
    DECLARE v_total_count INT;
    DECLARE v_current_index INT DEFAULT 0;
    DECLARE v_item_data JSON;
    DECLARE v_title VARCHAR(200);
    DECLARE v_price DECIMAL(10,2);
    DECLARE v_category_id INT;
    DECLARE CONTINUE HANDLER FOR SQLEXCEPTION
    BEGIN
        SET p_failed_count = p_failed_count + 1;
    END;
    
    SET p_success_count = 0;
    SET p_failed_count = 0;
    SET v_total_count = JSON_LENGTH(p_items_json);
    
    START TRANSACTION;
    
    WHILE v_current_index < v_total_count DO
        SET v_item_data = JSON_EXTRACT(p_items_json, CONCAT('$[', v_current_index, ']'));
        
        -- 提取零件数据
        SET v_title = JSON_UNQUOTE(JSON_EXTRACT(v_item_data, '$.title'));
        SET v_price = JSON_UNQUOTE(JSON_EXTRACT(v_item_data, '$.price'));
        SET v_category_id = JSON_UNQUOTE(JSON_EXTRACT(v_item_data, '$.category_id'));
        
        -- 插入零件
        INSERT INTO items (
            seller_id, title, price, category_id,
            status, created_at, updated_at, sync_version
        ) VALUES (
            p_seller_id, v_title, v_price, v_category_id,
            'available', NOW(), NOW(), 1
        );
        
        SET p_success_count = p_success_count + 1;
        SET v_current_index = v_current_index + 1;
    END WHILE;
    
    IF p_failed_count = 0 THEN
        COMMIT;
        SET p_status = 'success';
    ELSE
        ROLLBACK;
        SET p_status = 'partial_failure';
    END IF;
END$$

DELIMITER ;
```

```sql
DELIMITER $$

CREATE TRIGGER trg_users_after_insert
AFTER INSERT ON users
FOR EACH ROW
BEGIN
    DECLARE v_event_data JSON;
    
    -- 构造事件数据
    SET v_event_data = JSON_OBJECT(
        'event_id', UUID(),
        'event_type', 'INSERT',
        'table_name', 'users',
        'source_db', 'mysql',
        'record_id', NEW.id,
        'timestamp', NOW(),
        'data', JSON_OBJECT(
            'id', NEW.id,
            'username', NEW.username,
            'email', NEW.email,
            'password_hash', NEW.password_hash,
            'phone', NEW.phone,
            'is_active', NEW.is_active,
            'is_verified', NEW.is_verified,
            'is_banned', NEW.is_banned,
            'total_items_sold', NEW.total_items_sold,
            'total_items_purchased', NEW.total_items_purchased,
            'rating', NEW.rating,
            'created_at', NEW.created_at,
            'updated_at', NEW.updated_at,
            'sync_version', NEW.sync_version
        )
    );
    
    -- 记录到同步队列表（由后台进程处理）
    INSERT INTO sync_queue (
        event_type, table_name, record_id,
```

```sql
CREATE TRIGGER trg_items_after_update
AFTER UPDATE ON items
FOR EACH ROW
BEGIN
    DECLARE v_event_data JSON;
    DECLARE v_changes JSON;
    
    -- 检测变更字段
    SET v_changes = JSON_OBJECT();
    
    IF OLD.title != NEW.title THEN
        SET v_changes = JSON_SET(v_changes, '$.title', 
            JSON_OBJECT('old', OLD.title, 'new', NEW.title));
    END IF;
    
    IF OLD.price != NEW.price THEN
        SET v_changes = JSON_SET(v_changes, '$.price',

        SET v_changes = JSON_SET(v_changes, '$.status',

        'event_id', UUID(),
        'event_type', 'UPDATE',
        'table_name', 'items',
        'source_db', 'mysql',
        'record_id', NEW.id,
```

```sql
            UPDATE users

                updated_at = NOW()
            WHERE id = NEW.buyer_id;
            
            UPDATE users
            SET total_items_sold = total_items_sold + 1,
                updated_at = NOW()
            WHERE id = NEW.seller_id;
        END IF;
    END IF;
END$$
```

### 4.2.3 数据验证触发器

**功能说明：** 在数据插入或更新前进行业务规则验证。

#### 零件价格验证

```sql
DELIMITER $$

CREATE TRIGGER trg_items_validate_price
BEFORE INSERT ON items
FOR EACH ROW
BEGIN
    -- 验证价格必须为正数
    IF NEW.price <= 0 THEN
        SIGNAL SQLSTATE '45000'
        SET MESSAGE_TEXT = 'Item price must be greater than 0';
    END IF;
    
    -- 验证价格不能超过上限
    IF NEW.price > 100000 THEN
        SIGNAL SQLSTATE '45000'
        SET MESSAGE_TEXT = 'Item price cannot exceed 100,000';
    END IF;
    
    -- 自动设置默认值
    IF NEW.view_count IS NULL THEN
        SET NEW.view_count = 0;
    END IF;
```

```sql
CREATE FUNCTION fn_calculate_discount_price(
    p_original_price DECIMAL(10,2),

DETERMINISTIC
BEGIN
    DECLARE v_discount_price DECIMAL(10,2);
```

```sql
DELIMITER $$

CREATE FUNCTION fn_get_user_credit_level(p_user_id INT)
RETURNS VARCHAR(20)
READS SQL DATA
BEGIN
    DECLARE v_rating DECIMAL(3,2);
    DECLARE v_total_sales INT;
    DECLARE v_total_purchases INT;
    DECLARE v_is_verified BOOLEAN;
    DECLARE v_credit_level VARCHAR(20);
    
    -- 获取用户数据
    SELECT rating, total_items_sold, total_items_purchased, is_verified
    INTO v_rating, v_total_sales, v_total_purchases, v_is_verified
    FROM users
    WHERE id = p_user_id;
    
    -- 计算信用等级
    IF v_is_verified = FALSE THEN
        SET v_credit_level = 'Unverified';
    ELSEIF v_rating >= 4.5 AND (v_total_sales + v_total_purchases) >= 50 THEN
        SET v_credit_level = 'Diamond';
    ELSEIF v_rating >= 4.0 AND (v_total_sales + v_total_purchases) >= 30 THEN
        SET v_credit_level = 'Gold';
    ELSEIF v_rating >= 3.5 AND (v_total_sales + v_total_purchases) >= 10 THEN
        SET v_credit_level = 'Silver';
    ELSE
        SET v_credit_level = 'Bronze';
    END IF;
    
    RETURN v_credit_level;
END$$
```

#### 计算两地距离

```sql
DELIMITER $$

CREATE FUNCTION fn_calculate_distance(
    p_lat1 DECIMAL(10,8),
    p_lon1 DECIMAL(11,8),
    p_lat2 DECIMAL(10,8),
    p_lon2 DECIMAL(11,8)
)
RETURNS DECIMAL(10,2)
DETERMINISTIC
BEGIN
    DECLARE v_distance DECIMAL(10,2);
    DECLARE v_earth_radius INT DEFAULT 6371; -- 地球半径（公里）
    
    -- 使用Haversine公式计算距离
    SET v_distance = v_earth_radius * ACOS(
        COS(RADIANS(p_lat1)) * COS(RADIANS(p_lat2)) * 
        COS(RADIANS(p_lon2) - RADIANS(p_lon1)) + 
        SIN(RADIANS(p_lat1)) * SIN(RADIANS(p_lat2))
    );
    
    RETURN ROUND(v_distance, 2);
END$$

DELIMITER ;

-- 使用示例：查找附近的零件
SELECT 
    i.id, i.title, i.price, i.location,
    fn_calculate_distance(
        39.9042, 116.4074,  -- 用户位置（北京天安门）
        i.latitude, i.longitude
    ) AS distance_km
FROM items i
WHERE i.status = 'available'
HAVING distance_km <= 5  -- 5公里内
ORDER BY distance_km
LIMIT 20;
```

```sql
DELIMITER $$

CREATE PROCEDURE sp_expire_old_items(
    IN p_days_threshold INT,
    OUT p_expired_count INT
)
BEGIN
    DECLARE v_done INT DEFAULT FALSE;
    DECLARE v_item_id INT;
    DECLARE v_seller_id INT;
    DECLARE v_title VARCHAR(200);
    
    -- 声明游标
    DECLARE cur_old_items CURSOR FOR
        SELECT id, seller_id, title
        FROM items
        WHERE status = 'available'
        AND DATEDIFF(NOW(), created_at) > p_days_threshold;
    
    -- 声明游标结束处理
    DECLARE CONTINUE HANDLER FOR NOT FOUND SET v_done = TRUE;
    
    SET p_expired_count = 0;
    
    -- 打开游标
    OPEN cur_old_items;
    
    -- 循环处理
    read_loop: LOOP
        FETCH cur_old_items INTO v_item_id, v_seller_id, v_title;
        
        IF v_done THEN
            LEAVE read_loop;
        END IF;
        
        -- 更新零件状态
        UPDATE items
        SET status = 'expired',
            updated_at = NOW(),
            sync_version = sync_version + 1
        WHERE id = v_item_id;
        
        -- 发送通知给供应商
        INSERT INTO notifications (
            user_id, type, title, content,
            related_type, related_id, is_read, created_at
        ) VALUES (
            v_seller_id, 'item_expired', '零件已过期',
            CONCAT('您的零件《', v_title, '》已超过', p_days_threshold, '天未售出，已自动下架'),
            'item', v_item_id, FALSE, NOW()
        );
        
        SET p_expired_count = p_expired_count + 1;
    END LOOP;
    
    -- 关闭游标
    CLOSE cur_old_items;
END$$

DELIMITER ;
```

```sql
DELIMITER $$

CREATE PROCEDURE sp_calculate_user_activity_scores()
BEGIN
    DECLARE v_done INT DEFAULT FALSE;
    DECLARE v_user_id INT;
    DECLARE v_score INT;
    DECLARE v_items_count INT;
    DECLARE v_transactions_count INT;
    DECLARE v_messages_count INT;
    DECLARE v_last_login_days INT;
    
    -- 声明游标
    DECLARE cur_users CURSOR FOR
        SELECT id FROM users WHERE is_active = TRUE;
    
    DECLARE CONTINUE HANDLER FOR NOT FOUND SET v_done = TRUE;
    
    -- 打开游标
    OPEN cur_users;
    
    read_loop: LOOP
        FETCH cur_users INTO v_user_id;
        
        IF v_done THEN
            LEAVE read_loop;
        END IF;
        
        -- 计算各项指标
        -- 1. 发布零件数（最近30天）
        SELECT COUNT(*) INTO v_items_count
        FROM items
        WHERE seller_id = v_user_id
        AND created_at >= DATE_SUB(NOW(), INTERVAL 30 DAY);
        
        -- 2. 交易数（最近30天）
        SELECT COUNT(*) INTO v_transactions_count
        FROM transactions
        WHERE (buyer_id = v_user_id OR seller_id = v_user_id)
        AND created_at >= DATE_SUB(NOW(), INTERVAL 30 DAY);
        
        -- 3. 消息数（最近30天）
        SELECT COUNT(*) INTO v_messages_count
        FROM messages
        WHERE sender_id = v_user_id
        AND created_at >= DATE_SUB(NOW(), INTERVAL 30 DAY);
        
        -- 4. 最后登录天数
        SELECT DATEDIFF(NOW(), last_login_at) INTO v_last_login_days
        FROM users
        WHERE id = v_user_id;
        
        -- 计算综合分数
        SET v_score = (v_items_count * 10) + 
                      (v_transactions_count * 20) + 
                      (v_messages_count * 2) - 
                      (COALESCE(v_last_login_days, 30) * 5);
        
        -- 确保分数不为负
        IF v_score < 0 THEN
            SET v_score = 0;
        END IF;
        
        -- 更新用户活跃度分数
        UPDATE users
        SET activity_score = v_score,
            updated_at = NOW()
        WHERE id = v_user_id;
    END LOOP;
    
    CLOSE cur_users;
END$$

DELIMITER ;
```

## 4.4 事务管理

### 4.4.1 事务隔离级别

**测试不同隔离级别的效果：**

```sql
-- 1. READ UNCOMMITTED（读未提交）- 会出现脏读
-- 会话1
SET SESSION TRANSACTION ISOLATION LEVEL READ UNCOMMITTED;
START TRANSACTION;
UPDATE items SET price = 999.99 WHERE id = 1;
-- 暂不提交

-- 会话2
SET SESSION TRANSACTION ISOLATION LEVEL READ UNCOMMITTED;
START TRANSACTION;
SELECT price FROM items WHERE id = 1;  -- 能读到999.99（脏读）
COMMIT;

-- 会话1
ROLLBACK;  -- 回滚，price恢复原值
```

```sql
-- 2. READ COMMITTED（读已提交）- 避免脏读，但会出现不可重复读
-- 会话1
SET SESSION TRANSACTION ISOLATION LEVEL READ COMMITTED;
START TRANSACTION;
SELECT price FROM items WHERE id = 1;  -- 假设读到100.00

-- 会话2
START TRANSACTION;
UPDATE items SET price = 200.00 WHERE id = 1;
COMMIT;

-- 会话1
SELECT price FROM items WHERE id = 1;  -- 读到200.00（不可重复读）
COMMIT;
```

```sql
-- 3. REPEATABLE READ（可重复读）- MySQL默认级别
-- 会话1
SET SESSION TRANSACTION ISOLATION LEVEL REPEATABLE READ;
START TRANSACTION;
SELECT price FROM items WHERE id = 1;  -- 读到100.00

-- 会话2
START TRANSACTION;
UPDATE items SET price = 300.00 WHERE id = 1;
COMMIT;

-- 会话1
SELECT price FROM items WHERE id = 1;  -- 仍然读到100.00（可重复读）
COMMIT;

-- 会话1新事务
START TRANSACTION;
SELECT price FROM items WHERE id = 1;  -- 现在读到300.00
```

```sql
-- 4. SERIALIZABLE（串行化）- 最高隔离级别
-- 会话1
SET SESSION TRANSACTION ISOLATION LEVEL SERIALIZABLE;
START TRANSACTION;
SELECT * FROM items WHERE price < 100;

-- 会话2
START TRANSACTION;
INSERT INTO items (title, price, ...) VALUES ('New Item', 50, ...);
-- 会被阻塞，直到会话1提交
```

### 4.4.2 死锁检测与处理

**死锁场景模拟：**

```sql
-- 会话1
START TRANSACTION;
UPDATE items SET price = 100 WHERE id = 1;  -- 锁定item 1
-- 等待2秒
UPDATE items SET price = 200 WHERE id = 2;  -- 尝试锁定item 2

-- 会话2（同时执行）
START TRANSACTION;
UPDATE items SET price = 200 WHERE id = 2;  -- 锁定item 2
-- 等待2秒
UPDATE items SET price = 100 WHERE id = 1;  -- 尝试锁定item 1（死锁！）

-- MySQL会自动检测死锁并回滚其中一个事务
-- ERROR 1213 (40001): Deadlock found when trying to get lock
```

**死锁预防策略：**

```sql
-- 1. 按顺序访问资源
START TRANSACTION;
-- 总是按ID顺序更新，避免循环等待
UPDATE items SET price = 100 WHERE id = 1;
UPDATE items SET price = 200 WHERE id = 2;
UPDATE items SET price = 300 WHERE id = 3;
COMMIT;

-- 2. 使用SELECT ... FOR UPDATE提前锁定
START TRANSACTION;
-- 一次性锁定所有需要的记录
SELECT * FROM items WHERE id IN (1, 2, 3) FOR UPDATE;
-- 然后执行更新
UPDATE items SET price = price * 1.1 WHERE id IN (1, 2, 3);
COMMIT;

-- 3. 减小事务粒度
-- 不好的做法：长事务
START TRANSACTION;
UPDATE items SET price = 100 WHERE id = 1;
-- 执行大量其他操作...
SELECT SLEEP(10);
UPDATE items SET price = 200 WHERE id = 2;
COMMIT;

-- 好的做法：拆分为多个短事务
START TRANSACTION;
UPDATE items SET price = 100 WHERE id = 1;
COMMIT;

-- 其他操作...

START TRANSACTION;
UPDATE items SET price = 200 WHERE id = 2;
COMMIT;
```
